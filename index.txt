<script>
    window.onload = function() { window.IMP.init("imp50011546"); };

    function limitAndMove(el, max, nextId) {
        el.value = el.value.replace(/[^0-9]/g, '');
        if (el.value.length >= 2) {
            if (parseInt(el.value) > max) el.value = max;
            if (nextId) document.getElementById(nextId).focus();
        }
    }
    function toggleTime() {
        const dis = document.getElementById('noTime').checked;
        ['ampm', 'hour', 'min'].forEach(id => document.getElementById(id).disabled = dis);
    }
    function runAnalysis() {
        const btn = document.querySelector('.main-btn'); btn.innerText = "읽는 중...";
        fetch('/analyze', {
            method: 'POST', headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                name: document.getElementById('userName').value, gender: document.getElementById('userGender').value,
                birth: `${document.getElementById('year').value}-${document.getElementById('month').value}-${document.getElementById('day').value}`,
                time: document.getElementById('noTime').checked ? "모름" : `${document.getElementById('ampm').value} ${document.getElementById('hour').value}시 ${document.getElementById('min').value}분`
            })
        }).then(res => res.json()).then(data => {
            btn.innerText = "내 운세 확인하기"; document.getElementById('result').style.display = 'block';
            document.getElementById('freeText').innerText = data.summary; window.fullData = data.full;
        });
    }
    function startPayment() { document.getElementById('payModal').style.display = 'block'; }
    function requestPay() {
        window.IMP.request_pay({ pg: "kakaopay.TC0ONETIME", amount: 3900, name: "유도령 사주", merchant_uid: "yudoryung_" + new Date().getTime(), buyer_name: document.getElementById('userName').value }, 
        function (rsp) { if (rsp.success) { alert("복채 확인 완료!"); completePayment(); } else { alert("실패: " + rsp.error_msg); } });
    }
    function completePayment() {
        document.getElementById('payModal').style.display = 'none'; document.getElementById('paySection').style.display = 'none';
        document.getElementById('fullText').style.display = 'block'; document.getElementById('fullText').innerText = window.fullData;
    }
</script>
